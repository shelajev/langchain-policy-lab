FROM @BASE@ AS build
ENV UV_PYTHON_INSTALL_DIR=/opt/@NAME@/python
ENV UV_PROJECT_ENVIRONMENT=/opt/@NAME@/venv
ENV UV_LINK_MODE=copy
WORKDIR /opt/@NAME@/app
COPY app/ ./
RUN uv python install --no-bin @PYTHON@ && uv sync --locked --no-dev --no-editable --python @PYTHON@ --managed-python
RUN chown -R 0:0 /opt/@NAME@

FROM scratch
COPY --from=build /opt/@NAME@ /opt/@NAME@
ENV PATH="/opt/@NAME@/venv/bin"
